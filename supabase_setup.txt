-- TABLA 7: PROFILES (Perfiles y Créditos - NUEVA)

CREATE TABLE IF NOT EXISTS profiles (
    id UUID REFERENCES auth.users ON DELETE CASCADE PRIMARY KEY,
    credits INTEGER DEFAULT 500, -- Créditos iniciales
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Habilitar seguridad (RLS)
ALTER TABLE profiles ENABLE ROW LEVEL SECURITY;

-- Política: Los usuarios pueden ver su propio perfil
-- Eliminamos la política si existe para evitar errores de duplicado al correr el script
DROP POLICY IF EXISTS "Users can view own profile" ON profiles;
CREATE POLICY "Users can view own profile" ON profiles
    FOR SELECT USING (auth.uid() = id);

-- FUNCIÓN SEGURA: Deducción de créditos (RPC)
-- Esta función corre en el servidor y evita que el usuario se ponga créditos infinitos
CREATE OR REPLACE FUNCTION deduct_credits(amount_to_deduct INTEGER)
RETURNS INTEGER AS $$
DECLARE
    current_credits INTEGER;
    new_credits INTEGER;
BEGIN
    -- Obtener créditos actuales del usuario que llama a la función
    SELECT credits INTO current_credits FROM profiles WHERE id = auth.uid();
    
    IF current_credits IS NULL THEN
        RAISE EXCEPTION 'Usuario no encontrado';
    END IF;

    IF current_credits < amount_to_deduct THEN
        RAISE EXCEPTION 'Créditos insuficientes';
    END IF;

    new_credits := current_credits - amount_to_deduct;

    -- Actualizar créditos
    UPDATE profiles SET credits = new_credits WHERE id = auth.uid();

    RETURN new_credits;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- FUNCIÓN SEGURA: Inicialización de créditos (Self-Healing)
-- Permite crear el perfil de forma segura si falló el trigger inicial o es un usuario antiguo
CREATE OR REPLACE FUNCTION initialize_user_credits()
RETURNS void AS $$
BEGIN
  INSERT INTO public.profiles (id, credits)
  VALUES (auth.uid(), 500)
  ON CONFLICT (id) DO NOTHING;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- TRIGGER: Crear perfil automáticamente al registrarse
-- Cuando un usuario se crea en auth.users, le creamos su entrada en profiles con 500 créditos
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO public.profiles (id, credits)
  VALUES (new.id, 500);
  RETURN new;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Vincular trigger (si no existe ya)
DROP TRIGGER IF EXISTS on_auth_user_created ON auth.users;
CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE PROCEDURE public.handle_new_user();

-- MIGRACIÓN FINAL: ASEGURAR BORRADO EN CASCADA (ON DELETE CASCADE)
-- ==============================================================================
DO $$
BEGIN
    -- Sessions
    IF EXISTS (SELECT 1 FROM information_schema.table_constraints WHERE constraint_name = 'sessions_user_id_fkey' AND table_name = 'sessions') THEN
        ALTER TABLE sessions DROP CONSTRAINT sessions_user_id_fkey;
    END IF;
    ALTER TABLE sessions ADD CONSTRAINT sessions_user_id_fkey FOREIGN KEY (user_id) REFERENCES auth.users(id) ON DELETE CASCADE;
    
    -- Project Metrics (Cascade delete if session is deleted)
    IF EXISTS (SELECT 1 FROM information_schema.table_constraints WHERE constraint_name = 'project_metrics_session_id_fkey' AND table_name = 'project_metrics') THEN
        ALTER TABLE project_metrics DROP CONSTRAINT project_metrics_session_id_fkey;
    END IF;
    ALTER TABLE project_metrics ADD CONSTRAINT project_metrics_session_id_fkey FOREIGN KEY (session_id) REFERENCES sessions(id) ON DELETE CASCADE;

    -- Brand Kit
    IF EXISTS (SELECT 1 FROM information_schema.table_constraints WHERE constraint_name = 'brand_kit_user_id_fkey' AND table_name = 'brand_kit') THEN
        ALTER TABLE brand_kit DROP CONSTRAINT brand_kit_user_id_fkey;
    END IF;
    ALTER TABLE brand_kit ADD CONSTRAINT brand_kit_user_id_fkey FOREIGN KEY (user_id) REFERENCES auth.users(id) ON DELETE CASCADE;

    -- Landing Page Analyses
    IF EXISTS (SELECT 1 FROM information_schema.table_constraints WHERE constraint_name = 'landing_page_analyses_user_id_fkey' AND table_name = 'landing_page_analyses') THEN
        ALTER TABLE landing_page_analyses DROP CONSTRAINT landing_page_analyses_user_id_fkey;
    END IF;
    ALTER TABLE landing_page_analyses ADD CONSTRAINT landing_page_analyses_user_id_fkey FOREIGN KEY (user_id) REFERENCES auth.users(id) ON DELETE CASCADE;

    -- Social Accounts
    IF EXISTS (SELECT 1 FROM information_schema.table_constraints WHERE constraint_name = 'social_accounts_user_id_fkey' AND table_name = 'social_accounts') THEN
        ALTER TABLE social_accounts DROP CONSTRAINT social_accounts_user_id_fkey;
    END IF;
    ALTER TABLE social_accounts ADD CONSTRAINT social_accounts_user_id_fkey FOREIGN KEY (user_id) REFERENCES auth.users(id) ON DELETE CASCADE;
END $$;
